jobs:
  validate-p1-m3b:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
        with:
          fetch-depth: 0
      - uses: actions/setup-node@v5
        with:
          node-version: '24'
          cache: npm
          cache-dependency-path: package-lock.json
      - name: Install dependencies reproducibly
        run: npm ci
      - name: Typecheck P1-M3B service
        run: npx tsc -p tsconfig.p1-m3b-mlb-scientific-capture.json --noEmit
      - name: Run focused P1-M3B tests
        run: npx tsx --test server/mlb-p1-scientific-capture-service.test.ts
      - name: Run contract, ownership, security and route regressions
        run: |
          npx tsx --test \
            server/mlb-p1-scientific-capture-contract.test.ts \
            server/mlb-p1-pregame-readiness-contract.test.ts \
            server/mlb-ledger-ownership-store.test.ts \
            server/security-private-read.test.ts \
            server/routes-modularization.test.ts
      - name: Build production backend
        run: npm run build:backend
      - name: Run S5A real-save regression
        run: npm run test:s5a-e2e
      - name: Run complete S5B shadow regression
        run: npm run test:s5b-shadow
      - name: Verify authenticated append-only and safety boundaries
        shell: bash
        run: |
          set -euo pipefail
          service='server/mlb-p1-scientific-capture-service.ts'
          routes='server/mlb-p1-scientific-capture-routes.ts'
          security='server/security.ts'
          docs='docs/P1_M3B_MLB_SCIENTIFIC_CAPTURE_SERVICE.md'

          grep -F 'courtedge-p1-m3b-scientific-capture-service.v1' "$service"
          grep -F '/api/mlb/p1/v1/scientific-captures' "$routes"
          grep -F 'appendOwnedPrediction' "$service"
          grep -F 'ownerAuthority: "AUTHENTICATED_SESSION"' "$service"
          grep -F 'courtEdgeAuthenticated === true' "$routes"
          grep -F 'scientific-captures' "$security"
          grep -F 'P1-M3C' "$docs"
          grep -F 'realFinancialExposure = 0' "$docs"

          ! grep -E 'store\.appendPrediction|automaticBetPlacement:[[:space:]]*true|realFinancialExposure:[[:space:]]*[1-9]|automaticModelChangesAllowed:[[:space:]]*true|automaticPromotionAllowed:[[:space:]]*true' "$service"
          ! grep -E 'resolveRequestUserId\(|COURTEDGE_SYSTEM_OWNER_USER_ID' "$routes"
      - name: Diff hygiene and protected extension scope
        shell: bash
        run: |
          set -euo pipefail
          git fetch origin integration/p0-staging-secure --depth=1
          git diff --check origin/integration/p0-staging-secure...HEAD
          changed=$(git diff --name-only origin/integration/p0-staging-secure...HEAD | sort)

          allowed=$(printf '%s\n' \
            '.github/workflows/validate-p1-m3b-mlb-scientific-capture.yml' \
            '.github/workflows/validate-p1-m3d-interactive-economic-review.yml' \

            '.github/workflows/validate-p1-m4c-backend-economic-response.yml' \
            '.github/workflows/validate-p1-m5a-real-cohort-activation.yml' \
            '.github/workflows/validate-p1-m6a2-mlb-market-odds-universe.yml' \
            'docs/P1_M3D_INTERACTIVE_ECONOMIC_REVIEW.md' \
            'docs/P1_M3E_MLB_OPERATING_ENVELOPE_READONLY.md' \
            'docs/P1_PREMIUM_NO_ULTRA_PROSPECTIVE_READONLY.md' \
            'docs/P1_M3F3B5_VSTEAM_OPPOSING_TEAM_ID.md' \
            'docs/P1_M3F3B5B_STATCAST_CERTIFICATION.md' \
            'docs/P1_M4C_BACKEND_ECONOMIC_RESPONSE.md' \
            'docs/P1_M5A_REAL_COHORT_ACTIVATION.md' \
            'docs/P1_M6A2_MLB_MARKET_ODDS_UNIVERSE.md' \
            'server/mlb-market-odds-normalizer.test.ts' \
            'server/mlb-market-odds-normalizer.ts' \
            'server/mlb-market-universe-odds-routes.test.ts' \
            'server/mlb-market-universe-odds-routes.ts' \
            'server/mlb-p1-economic-review-routes.ts' \
            'server/mlb-p1-economic-review.test.ts' \
            'server/mlb-p1-economic-review.ts' \
            'server/mlb-p1-operating-envelope-routes.ts' \
            'server/mlb-premium-no-ultra-prospective-routes.ts' \
            'server/mlb-p1-real-cohort-activation.test.ts' \
            'server/mlb-p1-real-cohort-activation.ts' \
            'server/mlb-p1-scientific-capture-service.test.ts' \
            'server/mlb-p1-scientific-capture-service.ts' \
            'server/mlb-statcast-matchup-certifier.test.ts' \
            'server/mlb-statcast-matchup-certifier.ts' \

            'server/mlb-statcast-matchup-identity-routes.ts' \
            'server/mlb-statcast-matchup-vsteam-identity.test.ts' \
            'server/mlb-statcast-matchup-vsteam-identity.ts' \
            'server/route-contract.extensions.json' \
            'server/routes.ts' \
            'tsconfig.p1-m3d-interactive-economic-review.json' \
            'tsconfig.p1-m3e-operating-envelope.json' \
            'tsconfig.p1-premium-no-ultra-prospective-readonly.json' \
            'tsconfig.p1-m3f3b5-vsteam-identity.json' \
            'tsconfig.p1-m5a-real-cohort-activation.json' \
            'tsconfig.p1-m6a2-market-odds.json'             '.github/workflows/validate-p1-m3e5-live-frozen-envelope.yml' \
            'docs/P1_M3E5_MLB_LIVE_FROZEN_OPERATING_ENVELOPE.md' \
            'server/mlb-p1-operating-envelope-frozen-live.test.ts' \

            'server/mlb-p1-operating-envelope-frozen-routes.ts' \
            'tsconfig.p1-m3e5-live-frozen-envelope.json' \

          unexpected=$(comm -23 <(printf '%s\n' "$changed") <(printf '%s\n' "$allowed"))
          [[ -z "$unexpected" ]]

          if grep -Fxq 'server/mlb-p1-scientific-capture-service.ts' <<< "$changed" \
            || grep -Fxq 'server/mlb-p1-scientific-capture-service.test.ts' <<< "$changed"; then

            grep -Fxq 'server/mlb-p1-scientific-capture-service.test.ts' <<< "$changed"
          fi

            grep -Fxq 'server/mlb-p1-economic-review.ts' <<< "$changed"
            grep -Fxq 'server/mlb-p1-economic-review.test.ts' <<< "$changed"
            grep -Fxq 'docs/P1_M5A_REAL_COHORT_ACTIVATION.md' <<< "$changed"
            grep -Fxq 'tsconfig.p1-m5a-real-cohort-activation.json' <<< "$changed"
          fi

          ! grep -Eq '^(server/(security|mlb-ledger|mlb-p1-scientific-capture-contract|mlb-p1-scientific-capture-routes))' <<< "$unexpected"
